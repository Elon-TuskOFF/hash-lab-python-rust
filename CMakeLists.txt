cmake_minimum_required(VERSION 3.21)
project(HashLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
qt_add_executable(HashLab src/main.cpp)
target_link_libraries(HashLab PRIVATE Qt6::Widgets)
target_precompile_headers(HashLab PRIVATE <cmath>)
if(WIN32)
  set_target_properties(HashLab PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
